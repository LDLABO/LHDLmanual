　　32ビット乗算器の検証

　リスト162　機能実行譜
 　  

entity sim
output RESET;
output A[32];
output B[32];
output Y[32];
output READY;

bitr tc[8];

   part main(RESET,A,B,Y,READY)

   tc=tc+1;

   if (tc<5) RESET=1; endif
   if (tc==27) RESET=1; endif

   if (tc>25)
      A=12345;
      B=67890;
   else
      A=1234;
      B=5678;
   endif
