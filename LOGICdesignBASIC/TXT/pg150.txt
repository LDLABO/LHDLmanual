logicname LDL06E19A

entity main
input  RESET;
input  A[32];
input  B[32];
output Y[32];
output READY;

bitr   ra[64],rb[64];
bitr   acc[64];

   if (RESET)
      ra.0:31=A;    { 下32ビットに設置 }
      rb.32:63=B;   { 上32ビットに設置 }
   else
      ra.1:63=ra.0:62;   { 上桁移動 }
      rb.0:62=rb.1:63;   { 下桁移動 }
   endif

   if (RESET)
      acc=0;
   else
      if (rb.32)
         acc=acc+ra;     { 累算 }
      else
         acc=acc;
      endif
   endif

   Y=acc.0:31;

   if (rb.32:63==0) READY=1; endif

<font color="#336633"><b>　　32 ビット乗算器</b></font>
<hr color="#4F9177">

<p class="i">
本稿の
<font color="blue">「3.4 乗算器」</font>
を基に32 ビットの乗算器を作ります。
</p>

のようになります。

<hr color="#4F9177" size="15">
<font color="#336633"><b>　　32 ビット乗算器の論理</b></font>
<hr color="#4F9177">

<p class="i">
8 ビットの乗算器を32 ビットに拡張しただけで機能の追加
はしていません、実行の様子は
<font color="red">図5.93</font> 
のようになります。実
行時間は乗算する値によって決まりますが最大で32 CLK かか
ります。
</p>
